{{/* Returns the map pins as a slice, so both the rendered caption and the footer script can use them without computing them twice. One pin per post. Coordinates come from data/places.csv (generated by tools/places/geocoder.py and hand-checked); a post may also carry its own `coords: [lat, lng]` in front matter, which wins. A post with neither is simply absent from the map — a missing pin beats a pin in the wrong country. */}} {{- $pins := slice -}} {{- $pages := where .Site.RegularPages "Type" "posts" -}} {{- range $pages -}} {{- $slug := path.Base .File.Dir -}} {{- $lat := "" -}}{{- $lng := "" -}}{{- $place := "" -}} {{- range $.Site.Data.places -}} {{- if eq .slug $slug -}}{{- $lat = .lat -}}{{- $lng = .lng -}}{{- $place = .place -}}{{- end -}} {{- end -}} {{- if and (not $lat) .Params.coords -}} {{- $lat = index .Params.coords 0 -}}{{- $lng = index .Params.coords 1 -}} {{- $place = or .Params.where "" -}} {{- end -}} {{- if $lat -}} {{- $thumb := "" -}} {{- with .Params.thumbnail -}} {{- with resources.Get . -}}{{- $thumb = (.Resize "600x").RelPermalink -}}{{- end -}} {{- end -}} {{- $stats := dict -}} {{- with .Params.stats -}}{{- $stats = . -}}{{- end -}} {{- $pins = $pins | append (dict "slug" $slug "lat" (float $lat) "lng" (float $lng) "place" $place "title" .Title "url" .RelPermalink "thumb" $thumb "desc" (.Params.description | default "") "date" (time.Format "2 Jan 2006" .Date) "distance" (or (index $stats "distance") "") "elevation" (or (index $stats "elevation") "") "categories" (.Params.categories | default (slice))) -}} {{- end -}} {{- end -}} {{- return $pins -}}
